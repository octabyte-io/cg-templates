#!/bin/sh
# Runs once, when the database is first created (as in ckan/ckan-docker's
# postgresql/docker-entrypoint-initdb.d): creates CKAN's own database user and
# database, and the DataStore database with its read-only user.
set -e

psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "$POSTGRES_DB" <<-EOSQL
    CREATE ROLE ckan NOSUPERUSER CREATEDB CREATEROLE LOGIN PASSWORD '$CKAN_DB_PASSWORD';
    CREATE DATABASE ckan OWNER ckan ENCODING 'utf-8';
    CREATE ROLE datastore_ro NOSUPERUSER NOCREATEDB NOCREATEROLE LOGIN PASSWORD '$DATASTORE_RO_PASSWORD';
    CREATE DATABASE datastore OWNER ckan ENCODING 'utf-8';
EOSQL
