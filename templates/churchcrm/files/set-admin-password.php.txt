<?php
/*
 * Runs once at each start, after the app answers. ChurchCRM's installer creates the
 * account "Admin" with the public default password "changeme". The first time it
 * finds that untouched account, this replaces the password with the one Cloudgate
 * generated (ADMIN_PASSWORD). It never changes a password someone has already set.
 */

// The bcrypt hash of "changeme" that mysql/install/Install.sql ships for user "Admin".
const SHIPPED_HASH = '$2y$12$oBq/URJVlaV58kIVAN9oHuDenAzFz8Xe1GneeFdIWEtNdclehz6Hu';

mysqli_report(MYSQLI_REPORT_OFF);
$deadline = time() + 600;

while (true) {
    // A request to the app makes ChurchCRM install its tables into the empty database.
    @file_get_contents(getenv('CHURCHCRM_INTERNAL_URL'));

    $db = @new mysqli(
        getenv('CHURCHCRM_DB_HOST'),
        getenv('CHURCHCRM_DB_USER'),
        getenv('CHURCHCRM_DB_PASSWORD'),
        getenv('CHURCHCRM_DB_NAME')
    );
    if (!$db->connect_errno) {
        $result = @$db->query("SELECT usr_Password FROM user_usr WHERE usr_UserName = 'Admin'");
        if ($result) {
            $row = $result->fetch_assoc();
            if ($row === null || $row['usr_Password'] !== SHIPPED_HASH) {
                echo "The Admin password was already set; nothing to do.\n";
                exit(0);
            }
            $statement = $db->prepare(
                "UPDATE user_usr SET usr_Password = ?, usr_NeedPasswordChange = 0 "
                . "WHERE usr_UserName = 'Admin' AND usr_Password = ?"
            );
            $hash = password_hash(getenv('ADMIN_PASSWORD'), PASSWORD_DEFAULT);
            $shipped = SHIPPED_HASH;
            $statement->bind_param('ss', $hash, $shipped);
            if ($statement->execute()) {
                echo "Set the Admin password.\n";
                exit(0);
            }
            fwrite(STDERR, "Could not set the Admin password: {$statement->error}\n");
            exit(1);
        }
    }

    if (time() > $deadline) {
        fwrite(STDERR, "ChurchCRM's tables did not appear within 10 minutes.\n");
        exit(1);
    }
    echo "Waiting for ChurchCRM to install its tables...\n";
    sleep(5);
}
