#!/bin/sh
# Replaces the admin password DHIS2 starts with ("district") by ADMIN_PASSWORD,
# the way DHIS2's production example does (github.com/dhis2/docker-deployment):
# by writing a bcrypt hash of it into the database. It leaves the password alone
# once it is no longer "district", so a password changed in DHIS2 is kept.
# It runs in the background of the database container, at each start.
# The hashing runs in the "postgres" database, because pgcrypto would clash with
# functions DHIS2 creates in its own database.
set -eu
export PGUSER="$POSTGRES_USER"

# DHIS2 creates the admin account during its first start; wait for it.
i=0
until hash=$(psql -d dhis -tAc "SELECT password FROM userinfo WHERE username = 'admin'" 2>/dev/null) && [ -n "$hash" ]; do
  i=$((i + 1))
  [ "$i" -le 180 ] || { echo "admin-password: DHIS2 did not create its admin account within 30 minutes"; exit 1; }
  sleep 10
done

psql -d postgres -qc "SET client_min_messages TO warning; CREATE EXTENSION IF NOT EXISTS pgcrypto"
is_default=$(echo "SELECT crypt('district', :'h') = :'h';" | psql -d postgres -tA -v h="$hash")
if [ "$is_default" != "t" ]; then
  echo "admin-password: the admin password was already changed; leaving it alone."
  exit 0
fi

new_hash=$(echo "SELECT crypt(:'pw', gen_salt('bf', 10));" | psql -d postgres -tA -v pw="$ADMIN_PASSWORD")
echo "UPDATE userinfo SET password = :'h', passwordlastupdated = now() WHERE username = 'admin';" \
  | psql -d dhis -q -v ON_ERROR_STOP=1 -v h="$new_hash"
echo "admin-password: set the admin password."
