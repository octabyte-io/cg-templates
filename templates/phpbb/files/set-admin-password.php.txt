<?php
// Gives the phpBB administrator the password Cloudgate generated.
//
// phpBB's installer refuses passwords longer than 30 characters and Cloudgate generates 32,
// so the install (run just before this) uses the first 30. This then stores the full
// password, only while the stored one is still those first 30 characters, so a password
// the administrator has changed in phpBB is never overwritten. Runs at every start.

$password = (string) getenv('ADMIN_PASSWORD');
$username = (string) getenv('PHPBB_USERNAME');
if ($password === '' || $username === '') {
    fwrite(STDERR, "ADMIN_PASSWORD or PHPBB_USERNAME is empty\n");
    exit(1);
}
$installPassword = substr($password, 0, 30);
$table = (getenv('PHPBB_TABLE_PREFIX') ?: 'phpbb_') . 'users';

mysqli_report(MYSQLI_REPORT_OFF);
$db = new mysqli(
    getenv('PHPBB_DATABASE_HOST'),
    getenv('PHPBB_DATABASE_USER'),
    getenv('PHPBB_DATABASE_PASSWORD'),
    getenv('PHPBB_DATABASE_NAME')
);
if ($db->connect_errno) {
    fwrite(STDERR, 'Could not connect to the database: ' . $db->connect_error . "\n");
    exit(1);
}

// user_type 3 is a founder, the kind of account the installer creates.
$stmt = $db->prepare("SELECT user_id, user_password FROM `$table` WHERE username = ? AND user_type = 3 LIMIT 1");
$stmt->bind_param('s', $username);
$stmt->execute();
$row = $stmt->get_result()->fetch_assoc();
if (!$row) {
    echo "No administrator named $username; nothing to change.\n";
    exit(0);
}

if (password_verify($password, $row['user_password'])) {
    echo "The administrator password is already set.\n";
    exit(0);
}
if (strlen($password) <= 30 || !password_verify($installPassword, $row['user_password'])) {
    echo "The administrator password has been changed in phpBB; leaving it as it is.\n";
    exit(0);
}

// phpBB reads bcrypt ($2y$) hashes with its bcrypt_2y driver.
$hash = password_hash($password, PASSWORD_BCRYPT);
$now = time();
$update = $db->prepare("UPDATE `$table` SET user_password = ?, user_passchg = ? WHERE user_id = ?");
$update->bind_param('sii', $hash, $now, $row['user_id']);
if (!$update->execute()) {
    fwrite(STDERR, 'Could not set the administrator password: ' . $update->error . "\n");
    exit(1);
}
echo "Set the administrator password to the one Cloudgate generated.\n";
