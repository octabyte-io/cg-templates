# Replaces Mealie's built-in first administrator (changeme@example.com / MyPassword)
# with ADMIN_EMAIL and ADMIN_PASSWORD, once, through Mealie's own API. It runs in
# the background next to Mealie at each start and waits until Mealie answers.
#
# It acts only while that default account exists (Mealie reports it as its
# "first login"), so it never touches an administrator you have already set up,
# and it never sends a wrong password that could lock an account.
import json
import os
import sys
import time
import urllib.error
import urllib.parse
import urllib.request

BASE = os.environ.get("MEALIE_ADMIN_URL", "http://127.0.0.1:9000").rstrip("/")
EMAIL = os.environ["ADMIN_EMAIL"].strip()
PASSWORD = os.environ["ADMIN_PASSWORD"]
DEFAULT_EMAIL = "changeme@example.com"
DEFAULT_PASSWORD = "MyPassword"


def call(method, path, body=None, token=None, form=False):
    headers = {"Accept": "application/json"}
    data = None
    if body is not None:
        if form:
            data = urllib.parse.urlencode(body).encode()
            headers["Content-Type"] = "application/x-www-form-urlencoded"
        else:
            data = json.dumps(body).encode()
            headers["Content-Type"] = "application/json"
    if token:
        headers["Authorization"] = "Bearer " + token
    req = urllib.request.Request(BASE + path, data=data, headers=headers, method=method)
    try:
        with urllib.request.urlopen(req, timeout=30) as res:
            raw = res.read()
            return res.status, json.loads(raw) if raw else None
    except urllib.error.HTTPError as err:
        raw = err.read()
        try:
            return err.code, json.loads(raw)
        except ValueError:
            return err.code, raw.decode(errors="replace")


def login(email, password):
    status, body = call("POST", "/api/auth/token", {"username": email, "password": password}, form=True)
    if status == 200 and isinstance(body, dict) and body.get("access_token"):
        return body["access_token"]
    return None


def first_login():
    for _ in range(120):
        try:
            status, body = call("GET", "/api/app/about/startup-info")
            if status == 200 and isinstance(body, dict):
                return bool(body.get("isFirstLogin"))
        except OSError:
            pass
        time.sleep(5)
    sys.exit("mealie-admin: Mealie did not answer at " + BASE)


def main():
    if len(PASSWORD) < 8:
        sys.exit("mealie-admin: ADMIN_PASSWORD must have at least 8 characters.")
    if not first_login():
        print("mealie-admin: Mealie's default administrator is already replaced; nothing to do.")
        return

    token = login(DEFAULT_EMAIL, DEFAULT_PASSWORD)
    if token:
        status, body = call(
            "PUT",
            "/api/users/password",
            {"currentPassword": DEFAULT_PASSWORD, "newPassword": PASSWORD},
            token=token,
        )
        if status != 200:
            sys.exit("mealie-admin: could not set the administrator's password: %s %s" % (status, body))
    # Changing the password ends the session, so sign in again with the new one.
    # This also covers a previous run that set the password but not the email.
    token = login(DEFAULT_EMAIL, PASSWORD)
    if not token:
        sys.exit(
            "mealie-admin: the default administrator exists but neither password works; "
            "sign in to Mealie as changeme@example.com and change it there."
        )

    status, user = call("GET", "/api/users/self", token=token)
    if status != 200 or not isinstance(user, dict):
        sys.exit("mealie-admin: could not read the administrator: %s %s" % (status, user))
    fields = [
        "id", "username", "fullName", "email", "authMethod", "admin", "group", "household",
        "advanced", "canInvite", "canManage", "canManageHousehold", "canOrganize",
    ]
    update = {k: user[k] for k in fields if k in user}
    update["email"] = EMAIL
    if update.get("fullName") in (None, "", "Change Me"):
        update["fullName"] = "Admin"
    status, body = call("PUT", "/api/users/" + str(user["id"]), update, token=token)
    if status != 200:
        sys.exit("mealie-admin: could not set the administrator's email: %s %s" % (status, body))

    if not login(EMAIL, PASSWORD):
        sys.exit("mealie-admin: the administrator was updated but signing in with the new email failed.")
    print("mealie-admin: Mealie's administrator is now %s, with the password Cloudgate generated." % EMAIL)


main()
