<?php
// Replaces the default "admin" / "pointofsale" login with the password Cloudgate generated.
// Runs at every start after the migrations, but changes the password only while it is still
// "pointofsale", so a password the owner has changed in the app is never overwritten.

$password = (string) getenv('ADMIN_PASSWORD');
if ($password === '') {
    fwrite(STDERR, "ADMIN_PASSWORD is empty\n");
    exit(1);
}

mysqli_report(MYSQLI_REPORT_OFF);
$db = new mysqli(getenv('MYSQL_HOST_NAME'), getenv('MYSQL_USERNAME'), getenv('MYSQL_PASSWORD'), getenv('MYSQL_DB_NAME'));
if ($db->connect_errno) {
    fwrite(STDERR, 'Could not connect to the database: ' . $db->connect_error . "\n");
    exit(1);
}

$result = $db->query("SELECT person_id, password, hash_version FROM ospos_employees WHERE username = 'admin' LIMIT 1");
if ($result === false || $result->num_rows === 0) {
    echo "No employee named admin; nothing to change.\n";
    exit(0);
}

$row = $result->fetch_assoc();
$isDefault = ((string) $row['hash_version'] === '1')
    ? $row['password'] === md5('pointofsale')
    : password_verify('pointofsale', $row['password']);

if (!$isDefault) {
    echo "The admin password is no longer the default; leaving it as it is.\n";
    exit(0);
}

$hash = password_hash($password, PASSWORD_DEFAULT);
$stmt = $db->prepare('UPDATE ospos_employees SET password = ?, hash_version = 2 WHERE person_id = ?');
$stmt->bind_param('si', $hash, $row['person_id']);
if (!$stmt->execute()) {
    fwrite(STDERR, 'Could not set the admin password: ' . $stmt->error . "\n");
    exit(1);
}
echo "Set the admin password to the one Cloudgate generated.\n";
