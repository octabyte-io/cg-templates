# Runs once at each start, before Calibre-Web, in the same image. Every step
# only acts when it is still needed, so a later start changes nothing:
#   1. no library yet: creates an empty Calibre library in /books (metadata.db,
#      from Calibre's own schema in calibre-metadata.sql);
#   2. no settings yet: creates Calibre-Web's app.db and points it at /books,
#      with book uploads turned on;
#   3. the "admin" account still has the default password admin123: replaces
#      it with ADMIN_PASSWORD, so the well-known login never works.
import os
import sqlite3
import subprocess
import sys
import uuid

from werkzeug.security import check_password_hash, generate_password_hash

LIBRARY = "/books"
METADATA_DB = os.path.join(LIBRARY, "metadata.db")
APP_DB = "/config/app.db"
UID = int(os.environ.get("PUID", "1000"))
GID = int(os.environ.get("PGID", "1000"))


def create_library():
    if os.path.exists(METADATA_DB):
        print("Calibre library found in /books.")
        return
    print("Creating an empty Calibre library in /books...")
    with open("/calibre-metadata.sql", encoding="utf-8") as f:
        schema = f.read()
    conn = sqlite3.connect(METADATA_DB + ".new")
    conn.executescript(schema)
    conn.execute("INSERT INTO library_id (uuid) VALUES (?)", (str(uuid.uuid4()),))
    conn.commit()
    conn.close()
    os.rename(METADATA_DB + ".new", METADATA_DB)
    for path in (LIBRARY, METADATA_DB):
        os.chown(path, UID, GID)


def create_settings():
    if os.path.exists(APP_DB):
        print("Calibre-Web settings found.")
        return
    print("Creating the Calibre-Web settings...")
    # The same first-run step as the image's own start script: a dry run
    # creates app.db with the default settings and the admin account.
    env = dict(os.environ, CALIBRE_DBPATH="/config")
    subprocess.run([sys.executable, "cps.py", "-d"], cwd="/app/calibre-web", env=env,
                   stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, check=False)
    if not os.path.exists(APP_DB):
        sys.exit("Calibre-Web did not create app.db.")
    conn = sqlite3.connect(APP_DB)
    conn.execute(
        "UPDATE settings SET config_calibre_dir = ?, config_uploading = 1,"
        " config_kepubifypath = '/usr/bin/kepubify'"
        " WHERE config_calibre_dir IS NULL OR config_calibre_dir = ''",
        (LIBRARY,),
    )
    conn.commit()
    conn.close()


def replace_default_password():
    password = os.environ.get("ADMIN_PASSWORD", "")
    if not password:
        sys.exit("ADMIN_PASSWORD is not set.")
    conn = sqlite3.connect(APP_DB)
    row = conn.execute("SELECT id, password FROM user WHERE lower(name) = 'admin'").fetchone()
    if row and row[1] and check_password_hash(row[1], "admin123"):
        # Written directly, as Calibre-Web's own "-s" option would, but without
        # its password rules: the generated password is letters and digits.
        conn.execute("UPDATE user SET password = ? WHERE id = ?",
                     (generate_password_hash(password), row[0]))
        conn.commit()
        print("Replaced the default admin password with ADMIN_PASSWORD.")
    else:
        print("The admin password is not the default; left as it is.")
    conn.close()


create_library()
create_settings()
replace_default_password()
for name in os.listdir("/config"):
    os.chown(os.path.join("/config", name), UID, GID)
