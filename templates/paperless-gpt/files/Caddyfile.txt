# paperless-gpt has no sign-in of its own, so this small proxy asks for a
# username and password before anyone reaches it. Cloudgate's own proxy in
# front handles the domain and TLS. BASIC_AUTH_HASH is the bcrypt hash of
# ADMIN_PASSWORD, made when the container starts (see docker-compose.yml).
{
	admin off
}

:80 {
	basic_auth {
		admin {$BASIC_AUTH_HASH}
	}

	reverse_proxy paperless-gpt:8080
}
