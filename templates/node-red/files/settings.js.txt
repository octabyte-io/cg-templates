// Node-RED settings for Cloudgate, mounted as /data/settings.js.
//
// It starts from the default settings file shipped in the official image and
// changes only what Node-RED's security guide asks for
// (https://nodered.org/docs/user-guide/runtime/securing-node-red):
//   - adminAuth: the editor and admin API need a sign-in. The user is "admin"
//     and the password is the one Cloudgate generated (NODE_RED_ADMIN_PASSWORD),
//     stored here only as a bcrypt hash, as Node-RED expects.
//   - credentialSecret: the key that encrypts the passwords and tokens saved in
//     flows (flows_cred.json), generated by Cloudgate instead of by Node-RED.
const bcrypt = require("bcryptjs");
const settings = require("/usr/src/node-red/node_modules/node-red/settings.js");

const password = process.env.NODE_RED_ADMIN_PASSWORD;
if (!password) {
    throw new Error("NODE_RED_ADMIN_PASSWORD is not set: refusing to start Node-RED without a sign-in");
}

settings.adminAuth = {
    type: "credentials",
    users: [{
        username: process.env.NODE_RED_ADMIN_USER || "admin",
        password: bcrypt.hashSync(password, 10),
        permissions: "*"
    }]
};

if (process.env.NODE_RED_CREDENTIAL_SECRET) {
    settings.credentialSecret = process.env.NODE_RED_CREDENTIAL_SECRET;
}

module.exports = settings;
